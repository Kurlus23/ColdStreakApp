// ----------------------- Helper Functions -----------------------

void loadCredentials() {
  prefs.begin("flowguard", true);
  wifiSSID   = prefs.getString("ssid", "");
  wifiPass   = prefs.getString("pass", "");
  botToken   = prefs.getString("token", "");
  chatID     = prefs.getString("chat", "");
  isActivated = prefs.getBool("activated", false);
  prefs.end();
}

void saveCredentials() {
  prefs.begin("flowguard", false);
  prefs.putString("ssid", wifiSSID);
  prefs.putString("pass", wifiPass);
  prefs.putString("token", botToken);
  prefs.putString("chat", chatID);
  prefs.putBool("activated", isActivated);
  prefs.end();
}

bool checkActivation() {
  if (isActivated) return true;

  HTTPClient http;
  http.setTimeout(5000);
  http.begin(ACTIVATION_URL);
  int code = http.GET();
  http.end();

  if (code > 0) {
    isActivated = true;
    saveCredentials();
    Serial.println("Device activated successfully");
    return true;
  }
  Serial.println("Activation check failed");
  return false;
}

void sendTelegram(String message) {
  if (botToken.length() < 10 || chatID.length() < 3) return;
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;
  String url = "https://api.telegram.org/bot" + botToken +
               "/sendMessage?chat_id=" + chatID +
               "&text=" + message;
  http.begin(url);
  http.GET();
  http.end();
}
